// 临时真实探针：只有显式授权运行此用例才会调用收费上游。
func TestRealBudgetProbe(t *testing.T) {
	if os.Getenv("REAL_BUDGET_PROBE") != "1" {
		t.Skip("explicit real-call opt-in required")
	}
	root, err := filepath.Abs("../../docs/research/2026-10-20-token-budget-real-regression")
	if err != nil {
		t.Fatal("resolve report path failed")
	}
	if err = os.MkdirAll(root, 0700); err != nil {
		t.Fatal("create report failed")
	}
	f, err := os.OpenFile(filepath.Join(root, "results.jsonl"), os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
	if err != nil {
		t.Fatal("report already exists or unavailable; refusing duplicate paid calls")
	}
	defer f.Close()
	home, err := os.UserHomeDir()
	if err != nil {
		t.Fatal("home unavailable")
	}
	db, err := gorm.Open(sqlite.Open("file:"+filepath.Join(home, "Library/Application Support/@portunus/web/data/portunus.db")+"?mode=ro"), &gorm.Config{Logger: logger.Default.LogMode(logger.Silent)})
	if err != nil {
		t.Fatal("read configured routes failed")
	}
	type route struct {
		ID      int64
		Name    string
		Type    protocol.Provider
		BaseURL string
		Key     string
		Model   string
	}
	var routes []route
	err = db.Table("channels c").Select("c.id,c.name,c.type,c.base_url,c.key,m.name as model").Joins("join models m on m.channel_id=c.id").Where("(c.id=1 and m.name=?) or (c.id=3 and m.name=?)", "deepseek/deepseek-v4.1-flash", "deepseek-flash").Order("c.id").Scan(&routes).Error
	sqlDB, _ := db.DB()
	if sqlDB != nil {
		sqlDB.Close()
	}
	if err != nil || len(routes) != 2 {
		t.Fatal("expected two configured routes; no calls sent")
	}
	raw, err := os.ReadFile("../../docs/research/2026-10-20-reasoning-ab-comparison/cases.json")
	if err != nil {
		t.Fatal("load synthetic prompt failed")
	}
	var cases []struct {
		ID     string `json:"id"`
		Prompt string `json:"prompt"`
	}
	if json.Unmarshal(raw, &cases) != nil || len(cases) == 0 {
		t.Fatal("invalid prompt")
	}
	for _, route := range routes {
		if route.Type != protocol.ProviderOpenAI {
			t.Fatal("configured route protocol changed; refusing calls")
		}
		for _, kind := range []string{"codex", "claude_code"} {
			t.Run(fmt.Sprintf("channel_%d_%s", route.ID, kind), func(t *testing.T) {
				started := time.Now()
				row := map[string]any{"channel_id": route.ID, "route": route.Name, "base_url": route.BaseURL, "upstream_model": route.Model, "client": kind, "budget": 256, "task": cases[0].ID, "started_utc": started.UTC().Format(time.RFC3339Nano)}
				defer func() {
					row["duration_ms"] = time.Since(started).Milliseconds()
					b, _ := json.Marshal(row)
					f.Write(append(b, '\n'))
					f.Sync()
					t.Log(string(b))
				}()
				up, err := protocol.NewUpstream(route.Type, route.BaseURL, route.Key)
				if err != nil {
					t.Fatal("invalid upstream")
				}
				client := NewHTTPClient(shared.DefaultProxyConfig())
				client.Timeout = 90 * time.Second
				defer client.CloseIdleConnections()
				attempts := []map[string]any{}
				bridge := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
					b, _ := io.ReadAll(r.Body)
					var sent map[string]any
					json.Unmarshal(b, &sent)
					a := map[string]any{"max_tokens": sent["max_tokens"], "max_completion_tokens": sent["max_completion_tokens"], "reasoning_effort": sent["reasoning_effort"], "thinking": sent["thinking"], "model": sent["model"], "request_id": r.Header.Get("X-Request-Id")}
					attempts = append(attempts, a)
					// 严格限制探针的真实调用次数与预算，避免误跑产生额外费用。
					if len(attempts) > 2 || sent["model"] != route.Model || (sent["max_tokens"] != float64(256) && sent["max_completion_tokens"] != float64(256)) {
						a["error"] = "probe guard rejected request"
						w.WriteHeader(502)
						return
					}
					req, e := http.NewRequestWithContext(r.Context(), http.MethodPost, up.ChatURL(route.Model, true), bytes.NewReader(b))
					if e != nil {
						w.WriteHeader(502)
						return
					}
					req.Header = up.ChatHeaders()
					req.Header.Set("X-Request-Id", r.Header.Get("X-Request-Id"))
					resp, e := client.Do(req)
					if e != nil {
						a["error"] = "upstream transport failed"
						w.WriteHeader(502)
						return
					}
					defer resp.Body.Close()
					a["status"] = resp.StatusCode
					w.Header().Set("Content-Type", resp.Header.Get("Content-Type"))
					w.WriteHeader(resp.StatusCode)
					var captured bytes.Buffer
					buf := make([]byte, 16384)
					for {
						n, e := resp.Body.Read(buf)
						if n > 0 {
							captured.Write(buf[:n])
							w.Write(buf[:n])
							if fl, ok := w.(http.Flusher); ok {
								fl.Flush()
							}
						}
						if e != nil {
							if e != io.EOF {
								a["read_error"] = "upstream read failed"
							}
							break
						}
					}
					sc := bufio.NewScanner(bytes.NewReader(captured.Bytes()))
					sc.Buffer(make([]byte, 65536), 4*1024*1024)
					for sc.Scan() {
						line := sc.Text()
						if !strings.HasPrefix(line, "data:") {
							continue
						}
						data := strings.TrimSpace(strings.TrimPrefix(line, "data:"))
						if data == "[DONE]" {
							a["done"] = true
							continue
						}
						var ev map[string]any
						if json.Unmarshal([]byte(data), &ev) != nil {
							continue
						}
						if v := ev["usage"]; v != nil {
							a["usage"] = v
						}
						if v := ev["error"]; v != nil {
							a["stream_error"] = v
						}
						if cs, ok := ev["choices"].([]any); ok {
							for _, c := range cs {
								if cm, ok := c.(map[string]any); ok && cm["finish_reason"] != nil {
									a["finish_reason"] = cm["finish_reason"]
								}
							}
						}
					}
					if resp.StatusCode >= 400 {
						a["error_body"] = strings.ReplaceAll(captured.String(), route.Key, "[REDACTED]")
					}
				})
				pc := shared.DefaultProxyConfig()
				pc.RetryCount = 0
				router, key, _ := setupGatewayDeps(t, withCfg(pc), protocol.ProviderOpenAI, bridge)
				if shared.DB.Model(&model.Model{}).Where("name = ?", "upstream-model").Update("name", route.Model).Error != nil {
					t.Fatal("configure isolated model failed")
				}
				path := "/v1/responses"
				body := map[string]any{"model": "my-model", "stream": true, "max_output_tokens": 256, "reasoning": map[string]any{"effort": "high"}, "input": cases[0].Prompt}
				row["effort"] = "high"
				if kind == "claude_code" {
					path = "/v1/messages"
					row["effort"] = "max"
					body = map[string]any{"model": "my-model", "stream": true, "max_tokens": 256, "output_config": map[string]any{"effort": "max"}, "messages": []any{map[string]any{"role": "user", "content": cases[0].Prompt}}}
				}
				payload, _ := json.Marshal(body)
				w := doReq(t, router, path, string(payload), key)
				row["client_status"] = w.Code
				row["attempts"] = attempts
				terminal := []map[string]any{}
				sc := bufio.NewScanner(strings.NewReader(w.Body.String()))
				sc.Buffer(make([]byte, 65536), 4*1024*1024)
				for sc.Scan() {
					line := sc.Text()
					if !strings.HasPrefix(line, "data:") {
						continue
					}
					var ev map[string]any
					if json.Unmarshal([]byte(strings.TrimSpace(strings.TrimPrefix(line, "data:"))), &ev) != nil {
						continue
					}
					ty, _ := ev["type"].(string)
					if ty == "message_delta" || ty == "message_stop" || ty == "error" || ty == "response.completed" || ty == "response.incomplete" || ty == "response.failed" {
						meta := map[string]any{"type": ty}
						for _, k := range []string{"usage", "delta", "error"} {
							if v := ev[k]; v != nil {
								meta[k] = v
							}
						}
						if resp, ok := ev["response"].(map[string]any); ok {
							for _, k := range []string{"status", "usage", "incomplete_details", "error"} {
								if v := resp[k]; v != nil {
									meta[k] = v
								}
							}
						}
						terminal = append(terminal, meta)
					}
				}
				row["client_terminal_events"] = terminal
				passed := w.Code == 200 && len(attempts) == 1
				for _, a := range attempts {
					u, _ := a["usage"].(map[string]any)
					passed = passed && a["status"] == 200 && a["finish_reason"] == "length" && a["done"] == true && u["completion_tokens"] == float64(256) && a["max_tokens"] == float64(256) && a["max_completion_tokens"] == nil && a["reasoning_effort"] == row["effort"]
				}
				row["budget_verified"] = passed
				if !passed {
					t.Error("real budget verification failed; see sanitized report")
				}
			})
		}
	}
}
