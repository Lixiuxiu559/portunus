 def read_stdout():
  for line in p.stdout:
   try:channel.put(json.loads(line))

     # 执行真实的本地文件读取；随机值从未放入模型提示词。
     value=json.loads(fixture.read_text())[key];event['value']=value

    result['unexpected_server_requests'].append(method);send({'id':m['id'],'error':{'code':-32601,'message':'Unexpected server request refused by probe'}})

     item=params.get('item',{})

   print(json.dumps({'model':model,'turn':idx+1,'status':result['turns'][-1]['status'],'new_tool_calls':len(result['tool_calls'])-before,'passed':passed}),flush=True)
