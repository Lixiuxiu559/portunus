type route struct {
	ID                  int64
	Name                string
	Type                protocol.Provider
	BaseURL, Key, Model string
}
type call struct {
	Index               int `json:"index"`
	ID, Name, Arguments string
}
type requestObs struct {
	Ordinal        int              `json:"ordinal"`
	RequestID      string           `json:"request_id"`
	Status         int              `json:"status"`
	Finish         string           `json:"finish_reason"`
	Usage          any              `json:"usage"`
	Done           bool             `json:"done"`
	Text           string           `json:"text"`
	Reasoning      string           `json:"reasoning"`
	Tools          []*call          `json:"tool_calls"`
	History        []map[string]any `json:"history"`
	PairingValid   bool             `json:"pairing_valid"`
	InputToolNames []string         `json:"input_tool_names"`
	Error          string           `json:"error,omitempty"`
}
type session struct {
	mu            sync.Mutex
	ID            string         `json:"id"`
	ChannelID     int64          `json:"channel_id"`
	Model         string         `json:"model"`
	Budget        int            `json:"budget"`
	Requests      []*requestObs  `json:"requests"`
	Logs          []*shared.Log  `json:"gateway_logs"`
	Blocked       int            `json:"blocked_extra_calls"`
	HTTPCalls     int            `json:"http_calls"`
	ClientSummary map[string]any `json:"client_summary"`
	DriverExit    int            `json:"driver_exit"`
	IncomingTools []int          `json:"incoming_tool_counts"`
}
type guard struct {
	base   http.RoundTripper
	active *session
}

func history(messages []any) ([]map[string]any, bool) {
	var out []map[string]any
	valid := true
	pending := map[string]bool{}
	for _, raw := range messages {
		m, _ := raw.(map[string]any)
		role, _ := m["role"].(string)
		entry := map[string]any{"role": role}
		if role != "tool" && len(pending) > 0 {
			valid = false
		}
		if role == "assistant" {
			if tools, ok := m["tool_calls"].([]any); ok {
				var calls []map[string]any
				for _, v := range tools {
					tc, _ := v.(map[string]any)
					f, _ := tc["function"].(map[string]any)
					id, _ := tc["id"].(string)
					pending[id] = true
					calls = append(calls, map[string]any{"id": id, "name": f["name"], "arguments": f["arguments"]})
				}
				entry["tool_calls"] = calls
			}
		}
		if role == "tool" {
			id, _ := m["tool_call_id"].(string)
			if !pending[id] {
				valid = false
			}
			delete(pending, id)
			entry["tool_call_id"] = id
			entry["name"] = m["name"]
			entry["content"] = m["content"]
		}
		// 只记录结构及合成工具结果；不记录客户端自动加入的说明或真实业务正文。
		out = append(out, entry)
	}
	return out, valid && len(pending) == 0
}
func (g *guard) RoundTrip(req *http.Request) (*http.Response, error) {
	s := g.active
	raw, e := io.ReadAll(req.Body)
	if e != nil {
		return nil, e
	}
	req.Body.Close()
	var body map[string]any
	if json.Unmarshal(raw, &body) != nil {
		return nil, fmt.Errorf("probe invalid body")
	}
	s.mu.Lock()
	if len(s.Requests) >= 4 {
		s.Blocked++
		s.mu.Unlock()
		return nil, fmt.Errorf("probe max four paid calls per route")
	}
	if body["model"] != s.Model || body["max_tokens"] != float64(s.Budget) || body["max_completion_tokens"] != nil {
		s.mu.Unlock()
		return nil, fmt.Errorf("probe budget/model guard")
	}
	o := &requestObs{Ordinal: len(s.Requests) + 1, RequestID: req.Header.Get("X-Request-Id")}
	if messages, ok := body["messages"].([]any); ok {
		o.History, o.PairingValid = history(messages)
	}
	tools, _ := body["tools"].([]any)
	for _, raw := range tools {
		tool, _ := raw.(map[string]any)
		fn, _ := tool["function"].(map[string]any)
		if tool["type"] != "function" || fn["name"] != "read_probe_value" {
			s.mu.Unlock()
			return nil, fmt.Errorf("unexpected outbound tool")
		}
		o.InputToolNames = append(o.InputToolNames, fmt.Sprint(fn["name"]))
	}
	if len(o.InputToolNames) != 1 {
		s.mu.Unlock()
		return nil, fmt.Errorf("expected exactly one probe tool")
	}
	s.Requests = append(s.Requests, o)
	s.mu.Unlock()
	req.Body = io.NopCloser(bytes.NewReader(raw))
	resp, e := g.base.RoundTrip(req)
	if e != nil {
		s.mu.Lock()
		o.Error = "transport error"
		s.mu.Unlock()
		return nil, fmt.Errorf("upstream transport error")
	}
	s.mu.Lock()
	o.Status = resp.StatusCode
	s.mu.Unlock()
	resp.Body = &observedBody{ReadCloser: resp.Body, s: s, o: o}
	return resp, nil
}

type observedBody struct {
	io.ReadCloser
	s       *session
	o       *requestObs
	pending []byte
}

func (b *observedBody) Read(p []byte) (int, error) {
	n, e := b.ReadCloser.Read(p)
	if n > 0 {
		b.pending = append(b.pending, p[:n]...)
		for {
			idx := bytes.IndexByte(b.pending, '\n')
			if idx < 0 {
				break
			}
			line := strings.TrimSpace(string(b.pending[:idx]))
			b.pending = b.pending[idx+1:]
			b.observe(line)
		}
	}
	return n, e
}
func (b *observedBody) observe(line string) {
	if !strings.HasPrefix(line, "data:") {
		return
	}
	data := strings.TrimSpace(strings.TrimPrefix(line, "data:"))
	b.s.mu.Lock()
	defer b.s.mu.Unlock()
	o := b.o
	if data == "[DONE]" {
		o.Done = true
		return
	}
	var ev map[string]any
	if json.Unmarshal([]byte(data), &ev) != nil {
		return
	}
	if ev["usage"] != nil {
		o.Usage = ev["usage"]
	}
	choices, _ := ev["choices"].([]any)
	for _, raw := range choices {
		c, _ := raw.(map[string]any)
		if f, ok := c["finish_reason"].(string); ok && f != "" {
			o.Finish = f
		}
		d, _ := c["delta"].(map[string]any)
		if text, ok := d["content"].(string); ok {
			o.Text += text
		}
		if text, ok := d["reasoning_content"].(string); ok {
			o.Reasoning += text
		}
		tools, _ := d["tool_calls"].([]any)
		for _, raw := range tools {
			t, _ := raw.(map[string]any)
			index := int(t["index"].(float64))
			var target *call
			for _, candidate := range o.Tools {
				if candidate.Index == index {
					target = candidate
					break
				}
			}
			if target == nil {
				target = &call{Index: index}
				o.Tools = append(o.Tools, target)
			}
			if id, ok := t["id"].(string); ok && id != "" {
				target.ID = id
			}
			f, _ := t["function"].(map[string]any)
			if name, ok := f["name"].(string); ok && name != "" {
				target.Name = name
			}
			if args, ok := f["arguments"].(string); ok {
				target.Arguments += args
			}
		}
	}
}
func must(e error) {
	if e != nil {
		panic("probe setup failed: " + e.Error())
	}
}
func main() {
	if os.Getenv("PORTUNUS_REAL_TOOLS_PROBE") != "1" {
		panic("explicit real tool opt-in required")
	}
	report, e := filepath.Abs("docs/research/2026-10-20-responses-tools-multiturn")
	must(e)
	out, e := os.OpenFile(filepath.Join(report, "results.jsonl"), os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
	must(e)
	defer out.Close()
	home, e := os.UserHomeDir()
	must(e)
	db, e := gorm.Open(sqlite.Open("file:"+filepath.Join(home, "Library/Application Support/@portunus/web/data/portunus.db")+"?mode=ro"), &gorm.Config{Logger: logger.Default.LogMode(logger.Silent)})
	must(e)
	var routes []route
	must(db.Table("channels c").Select("c.id,c.name,c.type,c.base_url,c.key,m.name as model").Joins("join models m on m.channel_id=c.id").Where("(c.id=1 and m.name=?) or (c.id=3 and m.name=?)", "deepseek/deepseek-v4.1-flash", "deepseek-flash").Order("c.id").Scan(&routes).Error)
	sqlDB, e := db.DB()
	must(e)
	must(sqlDB.Close())
	if len(routes) != 2 {
		panic("routes unavailable")
	}
	shared.DB, e = gorm.Open(sqlite.Open(":memory:"), &gorm.Config{Logger: logger.Default.LogMode(logger.Silent)})
	must(e)
	must(shared.DB.AutoMigrate(&channel.Channel{}, &model.Model{}, &group.Group{}, &group.GroupItem{}))
	for _, r := range routes {
		if r.Type != protocol.ProviderOpenAI || r.Key == "" {
			panic("route changed")
		}
		ch := channel.Channel{ID: r.ID, Name: r.Name, Type: r.Type, BaseURL: r.BaseURL, Key: r.Key}
		must(shared.DB.Create(&ch).Error)
		m := model.Model{ChannelID: r.ID, Name: r.Model}
		must(shared.DB.Create(&m).Error)
		gr := group.Group{Name: fmt.Sprintf("probe%d", r.ID), Strategy: group.StrategyFailover}
		must(shared.DB.Create(&gr).Error)
		must(shared.DB.Create(&group.GroupItem{GroupID: gr.ID, ModelID: m.ID}).Error)
	}
	pc := shared.DefaultProxyConfig()
	pc.RetryCount = 0
	pc.FirstByteTimeoutSeconds = 45
	client := gateway.NewHTTPClient(pc)
	g := &guard{base: client.Transport}
	client.Transport = g
	tokenBytes := make([]byte, 16)
	_, e = rand.Read(tokenBytes)
	must(e)
	token := hex.EncodeToString(tokenBytes)
	gin.SetMode(gin.ReleaseMode)
	router := gin.New()
	router.Use(gin.Recovery())
	gateway.Register(router, gateway.Deps{Cfg: pc, Client: client, Breakers: gateway.NewBreakerStore(pc), Auth: func(k string) (int64, bool) { return 1, k == token }, LogWrite: func(l *shared.Log) {
		s := g.active
		s.mu.Lock()
		copy := *l
		s.Logs = append(s.Logs, &copy)
		s.mu.Unlock()
	}})
	srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		s := g.active
		s.mu.Lock()
		s.HTTPCalls++
		ordinal := s.HTTPCalls
		s.mu.Unlock()
		raw, e := io.ReadAll(r.Body)
		if e != nil {
			http.Error(w, "bad request", 400)
			return
		}
		var body map[string]any
		if json.Unmarshal(raw, &body) != nil {
			http.Error(w, "bad json", 400)
			return
		}
		tools, _ := body["tools"].([]any)
		selected := []any{}
		for _, raw := range tools {
			t, _ := raw.(map[string]any)
			if t["type"] == "function" && t["name"] == "read_probe_value" {
				selected = append(selected, t)
			}
		}
		s.mu.Lock()
		s.IncomingTools = append(s.IncomingTools, len(tools))
		s.mu.Unlock()
		if len(selected) != 1 {
			http.Error(w, "probe tool unavailable", 400)
			return
		}
		// 控制测试能力范围：仅保留已注册的只读动态函数，input 历史原样传递。
		body["tools"] = selected
		body["max_output_tokens"] = s.Budget
		raw, _ = json.Marshal(body)
		r.Body = io.NopCloser(bytes.NewReader(raw))
		r.ContentLength = int64(len(raw))
		tee := &teeWriter{ResponseWriter: w}
		router.ServeHTTP(tee, r)
		must(os.WriteFile(filepath.Join(report, fmt.Sprintf("%s-request-%d.response.sse", s.ID, ordinal)), tee.body.Bytes(), 0600))
	}))
	defer srv.Close()
	var sessions []*session
	for _, rt := range routes {
		s := &session{ID: fmt.Sprintf("probe%d", rt.ID), ChannelID: rt.ID, Model: rt.Model, Budget: 384}
		g.active = s
		fmt.Printf("starting %s max_paid_calls=4 budget=384\n", s.ID)
		ctx, cancel := context.WithTimeout(context.Background(), 210*time.Second)
		cmd := exec.CommandContext(ctx, "python3", "/tmp/portunus-tools-live/driver.py", "--url", srv.URL, "--model", s.ID, "--report", report)
		cmd.Env = append(os.Environ(), "PORTUNUS_PROBE_TOKEN="+token)
		cmd.Stdout = os.Stdout
		cmd.Stderr = os.Stderr
		e := cmd.Run()
		cancel()
		if e != nil {
			s.DriverExit = 1
		}
		b, e := os.ReadFile(filepath.Join(report, s.ID+".client-summary.json"))
		if e == nil {
			json.Unmarshal(b, &s.ClientSummary)
		}
		s.mu.Lock()
		encoded, _ := json.Marshal(s)
		for _, route := range routes {
			encoded = bytes.ReplaceAll(encoded, []byte(route.Key), []byte("[REDACTED]"))
		}
		_, e = out.Write(append(encoded, '\n'))
		must(e)
		must(out.Sync())
		fmt.Printf("finished %s calls=%d client_passed=%v\n", s.ID, len(s.Requests), s.ClientSummary["passed"])
		s.mu.Unlock()
		sessions = append(sessions, s)
	}
	summary := map[string]any{"created_utc": time.Now().UTC().Format(time.RFC3339), "max_total_paid_calls": 8, "sessions": sessions}
	encoded, _ := json.MarshalIndent(summary, "", "  ")
	must(os.WriteFile(filepath.Join(report, "summary.json"), encoded, 0600))
}

type teeWriter struct {
	http.ResponseWriter
	body bytes.Buffer
}

func (w *teeWriter) Write(b []byte) (int, error) { w.body.Write(b); return w.ResponseWriter.Write(b) }
func (w *teeWriter) Flush() {
	if f, ok := w.ResponseWriter.(http.Flusher); ok {
		f.Flush()
	}
}
