import pathlib,json,hashlib
root=pathlib.Path('docs/research/2026-10-20-responses-lifecycle-fix')
rows=[json.loads(x) for x in (root/'results.jsonl').read_text().splitlines()]
checks=[]
for r in rows:
 raw=(root/(r['id']+'.response.txt')).read_text()
 if r['stream']:
  events=[json.loads(s[6:].strip()) for s in raw.splitlines() if s.startswith('data: {')]
  final=events[-1]['response']
  seq=[e['sequence_number'] for e in events]
  assert seq==list(range(len(events))),r['id']
  items=[e['item'] for e in events if e['type']=='response.output_item.done']
  assert items==final['output'],r['id']
  assert raw.endswith('data: [DONE]\n\n'),r['id']
 else:
  events=[];final=json.loads(raw)
 output=final['output']
 text=''.join(c['text'] for item in output if item['type']=='message' for c in item.get('content',[]))
 reasoning=''.join(c['text'] for item in output if item['type']=='reasoning' for c in item.get('summary',[]))
 assert text==r['upstream_text'],r['id']
 assert reasoning==r['upstream_reasoning'],r['id']
 for item in output:
  for c in item.get('content',[]):
   if c['type']=='output_text':assert c['annotations']==[],r['id']
 u=r['upstream_usage'];cache=u.get('prompt_tokens_details',{}).get('cached_tokens',0)
 assert final['usage']['input_tokens']==u['prompt_tokens'],r['id']
 assert final['usage']['output_tokens']==u['completion_tokens'],r['id']
 assert final['usage']['input_tokens_details']['cached_tokens']==cache,r['id']
 log=r['gateway_log'];assert log['input_token']+log['cache_read_token']==u['prompt_tokens'],r['id']
 assert log['cache_read_token']==cache and log['output_token']==u['completion_tokens'],r['id']
 if 'codex_normal' in r['id']:
  events_cli=[json.loads(x) for x in (root/(r['id']+'.jsonl')).read_text().splitlines()]
  turns=[x for x in events_cli if x['type']=='turn.completed'];assert len(turns)==1,r['id']
  assert turns[0]['usage']['cached_input_tokens']==cache,r['id']
  assert r['cli_visible_text']==['OK'] and r['cli_exit']==0,r['id']
 else:
  assert final['status']=='incomplete' and final['incomplete_details']['reason']=='max_output_tokens',r['id']
  if r['id'].startswith('codex'):
   assert r['cli_exit']==1 and r['cli_truncation_recognized'],r['id']
 assert not r['cli_active_item_error'] and r['physical_upstream_calls']==1 and r['blocked_extra_attempts']==0,r['id']
 checks.append({'id':r['id'],'verified':True,'output_tokens':u['completion_tokens'],'cached_input_tokens':cache,'reasoning_chars':len(reasoning),'text_chars':len(text),'sequence_and_output_checked':bool(events)})
result={'all_checks_passed':True,'paid_calls':sum(r['physical_upstream_calls'] for r in rows),'input_tokens_including_cache':sum(r['upstream_usage']['prompt_tokens'] for r in rows),'cached_input_tokens':sum(r['upstream_usage'].get('prompt_tokens_details',{}).get('cached_tokens',0) for r in rows),'output_tokens':sum(r['upstream_usage']['completion_tokens'] for r in rows),'cases':checks,'source_sha256':{f:hashlib.sha256(pathlib.Path(f).read_bytes()).hexdigest() for f in ['backend/protocol/responses.go','backend/protocol/usage.go']}}
(root/'assessment.json').write_text(json.dumps(result,ensure_ascii=False,indent=2))
print(json.dumps({k:v for k,v in result.items() if k not in ['cases','source_sha256']},ensure_ascii=False))
