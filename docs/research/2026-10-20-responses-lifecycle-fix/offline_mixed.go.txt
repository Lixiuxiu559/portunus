package main
import("os";"encoding/json";"github.com/Lixiuxiu559/portunus/backend/protocol")
func main(){
 c,_:=protocol.NewStreamConverter(protocol.ProviderOpenAI,protocol.ProviderOpenAIResponses)
 f:=protocol.NewStreamFramer(protocol.ProviderOpenAIResponses,c)
 var all []byte
 for _,payload:=range []string{`{"id":"r1","model":"probe3","choices":[{"delta":{"content":"first"}}]}`, `{"choices":[{"delta":{"reasoning_content":"think"}}]}`, `{"choices":[{"delta":{"content":"OK"}}]}`,`{"id":"r1","choices":[{"delta":{},"finish_reason":"stop"}]}`,`{"choices":[],"usage":{"prompt_tokens":10,"completion_tokens":1,"total_tokens":11}}`}{parts,_:=f.ConvertPayload([]byte(payload));for _,p:=range parts{all=append(all,p...)}}
 parts,_:=f.FinishFrames();for _,p:=range parts{all=append(all,p...)}
 os.WriteFile("/tmp/portunus-responses-live-probe/mixed.sse",all,0600)
 c,_=protocol.NewStreamConverter(protocol.ProviderOpenAI,protocol.ProviderOpenAIResponses)
 payload:=[]byte(`{"id":"r1","choices":[{"delta":{"content":"x"},"finish_reason":"length"}],"usage":{"prompt_tokens":100,"completion_tokens":1,"total_tokens":101,"prompt_tokens_details":{"cached_tokens":80}}}`)
 c.Convert(payload);tail,_:=c.Finish();out:=map[string]any{"input_usage":c.Usage(),"terminal_events":[]json.RawMessage{}}
 for _,p:=range tail{out["terminal_events"]=append(out["terminal_events"].([]json.RawMessage),json.RawMessage(p))}
 b,_:=json.MarshalIndent(out,"","  ");os.WriteFile("/tmp/portunus-responses-live-probe/cache-offline.json",b,0600)
}
