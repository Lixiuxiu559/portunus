import http.server,threading,subprocess,os,json,pathlib,tempfile
report=pathlib.Path('docs/research/2026-10-20-responses-lifecycle-fix').resolve()
current=pathlib.Path('/tmp/portunus-responses-live-probe/current.sse').read_bytes()
item={'id':'msg_1','type':'message','role':'assistant','status':'completed','content':[{'type':'output_text','text':'OK','annotations':[]}]}
part={'type':'output_text','text':'','annotations':[]}
response={'id':'r1','object':'response','status':'completed','model':'probe3','output':[item],'usage':{'input_tokens':10,'output_tokens':1,'total_tokens':11,'input_tokens_details':{'cached_tokens':0}}}
events=[
 {'type':'response.created','response':dict(response,status='in_progress',output=[])},
 {'type':'response.output_item.added','output_index':0,'item':dict(item,status='in_progress',content=[])},
 {'type':'response.content_part.added','item_id':'msg_1','output_index':0,'content_index':0,'part':part},
 {'type':'response.output_text.delta','item_id':'msg_1','output_index':0,'content_index':0,'delta':'OK'},
 {'type':'response.output_text.done','item_id':'msg_1','output_index':0,'content_index':0,'text':'OK'},
 {'type':'response.content_part.done','item_id':'msg_1','output_index':0,'content_index':0,'part':item['content'][0]},
 {'type':'response.output_item.done','output_index':0,'item':item},
 {'type':'response.completed','response':response},
]
for n,e in enumerate(events):e['sequence_number']=n
control=(''.join('data: '+json.dumps(e)+'\n\n' for e in events)+'data: [DONE]\n\n').encode()
state={'mode':'current','calls':0}
class Handler(http.server.BaseHTTPRequestHandler):
 def log_message(self,*a):pass
 def do_POST(self):
  self.rfile.read(int(self.headers.get('Content-Length','0')))
  state['calls']+=1
  data=current if state['mode']=='current' else control
  self.send_response(200);self.send_header('Content-Type','text/event-stream');self.send_header('Content-Length',str(len(data)));self.end_headers();self.wfile.write(data)
server=http.server.ThreadingHTTPServer(('127.0.0.1',0),Handler)
thread=threading.Thread(target=server.serve_forever,daemon=True);thread.start()
rows=[]
try:
 for mode in ['current','control']:
  state.update(mode=mode,calls=0)
  with tempfile.TemporaryDirectory(prefix='portunus-codex-offline-',dir='/tmp') as work:
   args=['codex','exec','--ignore-user-config','--ignore-rules','--ephemeral','--skip-git-repo-check','--sandbox','read-only','--color','never','--json','-C',work,'-m','probe3']
   cfg={'model_provider':'portunus_offline','model_providers.portunus_offline.name':'offline-only','model_providers.portunus_offline.base_url':f'http://127.0.0.1:{server.server_port}/v1','model_providers.portunus_offline.env_key':'PORTUNUS_PROBE_TOKEN','model_providers.portunus_offline.wire_api':'responses','model_providers.portunus_offline.requires_openai_auth':False,'model_providers.portunus_offline.supports_websockets':False,'model_providers.portunus_offline.request_max_retries':0,'model_providers.portunus_offline.stream_max_retries':0,'model_reasoning_effort':'none','web_search':'disabled'}
   for k,v in cfg.items():args+=['-c',k+'='+json.dumps(v)]
   args+=['不要使用工具，只回复 OK。']
   proc=subprocess.run(args,input=b'',stdout=subprocess.PIPE,stderr=subprocess.PIPE,timeout=45,env=dict(os.environ,PORTUNUS_PROBE_TOKEN='offline-dummy-token'))
   stdout=proc.stdout.decode();stderr=proc.stderr.decode()
   (report/f'offline_{mode}.jsonl').write_text(stdout)
   (report/f'offline_{mode}.stderr.txt').write_text(stderr)
   parsed=[json.loads(x) for x in stdout.splitlines() if x.startswith('{')]
   texts=[e.get('item',{}).get('text','') for e in parsed if e.get('item',{}).get('type')=='agent_message']
   rows.append({'mode':mode,'exit_code':proc.returncode,'local_requests':state['calls'],'agent_text':texts,'active_item_error':'OutputTextDelta without active item' in stderr,'paid_calls':0})
finally:server.shutdown();server.server_close()
(report/'offline_lifecycle_summary.json').write_text(json.dumps(rows,ensure_ascii=False,indent=2))
(report/'offline_current.sse').write_bytes(current)
(report/'offline_control.sse').write_bytes(control)
print(json.dumps(rows,ensure_ascii=False))
